## Code

Fences go through Shiki with Vitesse Light and Vitesse Dark. The language tag on the fence picks the grammar.

### TypeScript

```ts
type Result<T> = { ok: true; value: T } | { ok: false; error: string };

const read = async (slug: string): Promise<Result<string>> => {
  if (!/^[a-z0-9-]+$/.test(slug)) {
    return { ok: false, error: 'bad slug' };
  }

  return { ok: true, value: slug };
};
```

### JavaScript

```js
export const fenceLanguage = (className) => {
  if (!className) {
    return 'plaintext';
  }

  const marker = 'language-';
  const index = className.indexOf(marker);
  return index === -1 ? 'plaintext' : className.slice(index + marker.length);
};
```

### Python

```python
from dataclasses import dataclass


@dataclass(frozen=True)
class Post:
    slug: str
    title: str
    date: str


def newest(posts: list[Post]) -> Post:
    return max(posts, key=lambda post: post.date)
```

### Rust

```rust
pub enum Result<T> {
    Ok(T),
    Err(String),
}

pub fn read(slug: &str) -> Result<String> {
    if slug.is_empty() {
        return Result::Err("bad slug".into());
    }

    Result::Ok(slug.to_string())
}
```

### Go

```go
package blog

type Post struct {
    Slug  string
    Title string
    Date  string
}

func Newest(posts []Post) Post {
    best := posts[0]
    for _, post := range posts[1:] {
        if post.Date > best.Date {
            best = post
        }
    }
    return best
}
```
